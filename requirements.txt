# Everything this agent needs beyond the standard library, pinned so an image
# built today is the image that was tested.
#
# Adding to this list is a real cost: the image build then needs a package
# index, and every dependency is a surface the successor has to keep working.
# The second experiment's record is unambiguous about which additions paid
# their way in exams and which rode along as fossils, and this list is the
# part that did.

# The official OpenAI client: the agent talks to any endpoint serving the
# Responses API through it, and gives the model its tools in the API's own
# function-calling format.
openai==3.3.1

# The same pytest the orchestrator's checker grades with, so a probe can run a
# task's tests instead of reasoning about them, a generator can prove its
# bundle before handing it in, and this agent's own tests run in its image.
pytest==8.4.1

# Native parsers for the PDFs and Office files that arrive as task materials.
# read_document was the one evolved reader exams actually called.
pypdf==5.9.0
python-docx==1.2.0
openpyxl==3.1.5
python-pptx==1.0.2
